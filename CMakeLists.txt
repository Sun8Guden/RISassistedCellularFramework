cmake_minimum_required(VERSION 3.20)
project(risgeo VERSION 1.0.0 DESCRIPTION "Semi-analytical performance engine for RIS-assisted cellular networks" LANGUAGES CXX)

option(RISGEO_USE_OPENMP "Parallelise the adaptive cubature with OpenMP when available" ON)
option(RISGEO_BUILD_EXAMPLES "Build the examples" ON)
option(RISGEO_BUILD_TESTS "Build the regression tests" ON)
option(RISGEO_BUILD_BENCHMARKS "Build the benchmarks" ON)
option(RISGEO_BUILD_VALIDATION "Build the Monte Carlo validation" ON)
option(RISGEO_BUILD_PAPER "Build the research drivers in paper/" OFF)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

# ---- Header-only library ----
add_library(risgeo INTERFACE)
add_library(risgeo::risgeo ALIAS risgeo)
target_include_directories(risgeo INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
target_include_directories(risgeo SYSTEM INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/third_party>)
target_compile_features(risgeo INTERFACE cxx_std_20)

if(RISGEO_USE_OPENMP)
  if(APPLE AND NOT DEFINED OpenMP_ROOT)
    # Apple Clang ships without OpenMP; use Homebrew's libomp when it is installed.
    execute_process(COMMAND brew --prefix libomp OUTPUT_VARIABLE _libomp_prefix
                    OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
    if(_libomp_prefix)
      set(OpenMP_ROOT "${_libomp_prefix}")
    endif()
  endif()
  find_package(OpenMP)
  if(OpenMP_CXX_FOUND)
    target_link_libraries(risgeo INTERFACE OpenMP::OpenMP_CXX)
  else()
    message(STATUS "OpenMP not found: the adaptive cubature runs serially")
  endif()
endif()

# ---- Programs ----
function(risgeo_add_program source)
  get_filename_component(absolute "${source}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
  file(RELATIVE_PATH relative "${CMAKE_CURRENT_SOURCE_DIR}" "${absolute}")
  get_filename_component(directory "${relative}" DIRECTORY)
  get_filename_component(name "${relative}" NAME_WE)
  if(directory MATCHES "^paper/")
    string(REPLACE "/" "_" prefix "${directory}")
    set(name "${prefix}_${name}")
  endif()
  add_executable(${name} ${source})
  target_link_libraries(${name} PRIVATE risgeo::risgeo)
  if(NOT directory MATCHES "^paper/")
    target_compile_options(${name} PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang,AppleClang>:-Wall -Wextra>)
  endif()
  set_target_properties(${name} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/${directory}")
endfunction()

if(RISGEO_BUILD_TESTS)
  enable_testing()
endif()

if(RISGEO_BUILD_EXAMPLES)
  file(GLOB example_sources CONFIGURE_DEPENDS examples/*.cpp)
  foreach(source ${example_sources})
    risgeo_add_program(${source})
  endforeach()
  if(RISGEO_BUILD_TESTS)
    add_test(NAME option_pricing COMMAND OptionPricingExample)
    set_tests_properties(option_pricing PROPERTIES LABELS validation)
  endif()
endif()

if(RISGEO_BUILD_TESTS)
  risgeo_add_program(tests/regression_test.cpp)
  add_test(NAME regression COMMAND regression_test)
endif()

if(RISGEO_BUILD_BENCHMARKS)
  risgeo_add_program(benchmarks/cubature_benchmark.cpp)
endif()

if(RISGEO_BUILD_VALIDATION)
  risgeo_add_program(validation/monte_carlo.cpp)
  if(RISGEO_BUILD_TESTS)
    add_test(NAME monte_carlo_validation COMMAND monte_carlo 400000)
    set_tests_properties(monte_carlo_validation PROPERTIES LABELS validation)
  endif()
endif()

if(RISGEO_BUILD_PAPER)
  file(GLOB_RECURSE paper_sources CONFIGURE_DEPENDS paper/*.cpp)
  foreach(source ${paper_sources})
    risgeo_add_program(${source})
  endforeach()
endif()
